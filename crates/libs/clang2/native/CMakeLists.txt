cmake_minimum_required(VERSION 3.20)
project(windows_clang_layout LANGUAGES C CXX)
find_package(LLVM 22.1.8 EXACT REQUIRED CONFIG)
find_package(Clang 22.1.8 EXACT REQUIRED CONFIG)
file(TO_CMAKE_PATH "${LIBCLANG_SOURCE}" LIBCLANG_SOURCE)
file(READ "${LIBCLANG_SOURCE}/CXCursor.cpp" cursor_source)
string(REGEX MATCH
    "const Decl \\*cxcursor::getCursorDecl\\(CXCursor Cursor\\) \\{[^}]+\\}"
    cursor_accessor "${cursor_source}")
if(NOT cursor_accessor)
    message(FATAL_ERROR "Pinned compiler cursor accessor is unavailable")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cursor_decl.inc"
    "namespace clang {\n${cursor_accessor}\n}\n")
add_library(windows_clang_layout SHARED layout.cpp)
target_compile_features(windows_clang_layout PRIVATE cxx_std_20)
target_include_directories(windows_clang_layout SYSTEM PRIVATE
    ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS} "${LIBCLANG_SOURCE}")
target_include_directories(windows_clang_layout PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(windows_clang_layout PRIVATE NDEBUG)
get_target_property(parser_image libclang IMPORTED_LOCATION_RELEASE)
file(SHA256 "${parser_image}" parser_sha256)
target_compile_definitions(windows_clang_layout PRIVATE
    WINDOWS_CLANG_PARSER_SHA256="${parser_sha256}")
target_compile_options(windows_clang_layout PRIVATE /GR- /Gy /W4 /WX)
target_link_options(windows_clang_layout PRIVATE /OPT:REF)
target_link_libraries(windows_clang_layout PRIVATE clangAST clangLex libclang)
set_property(TARGET windows_clang_layout PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
foreach(case IN ITEMS bad_abi bad_hash)
    add_library(windows_clang_layout_${case} SHARED mismatch.cpp)
    target_include_directories(windows_clang_layout_${case} SYSTEM PRIVATE ${CLANG_INCLUDE_DIRS})
    target_link_libraries(windows_clang_layout_${case} PRIVATE libclang)
    set_property(TARGET windows_clang_layout_${case} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endforeach()
target_compile_definitions(windows_clang_layout_bad_abi PRIVATE TEST_ABI=5)
target_compile_definitions(windows_clang_layout_bad_hash PRIVATE TEST_ABI=4)
